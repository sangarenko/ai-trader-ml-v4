    16→  C12: Supertrend — proper recursive calculation
    17→  M1: Donchian — exclude current bar

    19→  M3: Returns — divide by previous close
    20→  M4: RSI — Wilder smoothing
    21→  M5: Sortino — downside deviation
    22→  M6: Sortino annualization — 78 bars/day
    23→  M8: Balance — multiplicative compounding
    24→  M9: Exit not on entry bar
    25→  M11: BB std — ddof=1
    26→  M12: All 22 strategies implemented
    27→"""
    28→import numpy as np
    29→import time
    30→from typing import Dict, Tuple, List
    31→
    32→
    33→# ─── Pre-compute indicators (ONCE per ticker) ───────────────────────────────
    34→
    35→def precompute_indicators(opens: np.ndarray, closes: np.ndarray, highs: np.ndarray,
    36→                          lows: np.ndarray, volumes: np.ndarray) -> Dict[str, np.ndarray]:
    37→    """Compute all indicators ONCE. Returns dict of numpy arrays.
    38→    
    39→    All arrays are same length as input. Warmup period (first 50 bars) = 0.
    40→    """
    41→    n = len(closes)
    42→    ind = {}
    43→    
    44→    # Helper: proper rolling mean (no wraparound)
    45→    def rolling_mean(arr, w):
    46→        ret = np.cumsum(arr, dtype=float)
    47→        result = np.empty(n)
    48→        result[:w] = ret[:w] / np.arange(1, min(w, n) + 1)[:n]
    49→        if n > w:
    50→            result[w:] = (ret[w:] - ret[:-w]) / w
    51→        return result
    52→    
    53→    # Helper: rolling std with ddof=1
    54→    def rolling_std(arr, w):
    55→        mean = rolling_mean(arr, w)
    56→        sq_diff = (arr - mean) ** 2
    57→        return np.sqrt(rolling_mean(sq_diff, w) * w / max(w - 1, 1))
    58→    
    59→    # Helper: Wilder's smoothing (EMA with alpha = 1/period)
    60→    def wilder_smooth(arr, period):
    61→        alpha = 1.0 / period
    62→        result = np.empty(n)
    63→        result[0] = arr[0]
    64→        for i in range(1, n):
    65→            result[i] = alpha * arr[i] + (1 - alpha) * result[i - 1]
    66→        return result
    67→    
    68→    # Helper: EMA
    69→    def ema(arr, period):
    70→        alpha = 2.0 / (period + 1)
    71→        result = np.empty(n)
    72→        result[0] = arr[0]
    73→        for i in range(1, n):
    74→            result[i] = alpha * arr[i] + (1 - alpha) * result[i - 1]
    75→        return result
    76→    
    77→    # Returns — divide by PREVIOUS close (fix M3)
    78→    prev_close = np.empty(n)
    79→    prev_close[0] = closes[0]
    80→    prev_close[1:] = closes[:-1]
    81→    
    82→    ind["ret_1"] = (closes - prev_close) / (prev_close + 1e-10)
    83→    ind["ret_5"] = np.zeros(n)
    84→    if n > 5:
    85→        ind["ret_5"][5:] = (closes[5:] - closes[:-5]) / (closes[:-5] + 1e-10)
    86→    
    87→    # SMAs
    88→    ind["sma5"] = rolling_mean(closes, 5)
    89→    ind["sma14"] = rolling_mean(closes, 14)
    90→    ind["sma20"] = rolling_mean(closes, 20)
    91→    ind["sma50"] = rolling_mean(closes, 50) if n >= 50 else np.full(n, closes.mean())
    92→    
    93→    # RSI(14) — Wilder smoothing (fix M4)
    94→    deltas = np.diff(closes, prepend=closes[0])
    95→    gains = np.where(deltas > 0, deltas, 0.0)
    96→    losses = np.where(deltas < 0, -deltas, 0.0)
    97→    avg_gain = wilder_smooth(gains, 14)
    98→    avg_loss = wilder_smooth(losses, 14)
    99→    rs = avg_gain / (avg_loss + 1e-10)
   100→    ind["rsi14"] = 100 - 100 / (1 + rs)
   101→    
   102→    # RSI(2)
   103→    avg_gain2 = wilder_smooth(gains, 2)
   104→    avg_loss2 = wilder_smooth(losses, 2)
   105→    rs2 = avg_gain2 / (avg_loss2 + 1e-10)
   106→    ind["rsi2"] = 100 - 100 / (1 + rs2)
   107→    
   108→    # Bollinger Bands — ddof=1 (fix M11)
   109→    sma20 = ind["sma20"]
   110→    std20 = rolling_std(closes, 20)
   111→    ind["bb_upper"] = sma20 + 2 * std20
   112→    ind["bb_lower"] = sma20 - 2 * std20
   113→    ind["bb_width"] = (4 * std20) / (sma20 + 1e-10)
   114→    
   115→    # ATR (14) — Wilder
   116→    tr = np.maximum(highs - lows, np.maximum(
   117→        np.abs(highs - prev_close),
   118→        np.abs(lows - prev_close)
   119→    ))
   120→    ind["atr"] = wilder_smooth(tr, 14)
   121→    ind["atr_pct"] = ind["atr"] / (closes + 1e-10)
   122→    
   123→    # Volume ratio
   124→    vol_avg = rolling_mean(volumes.astype(float), 20)
   125→    ind["vol_ratio"] = volumes / (vol_avg + 1e-10)
   126→    
   127→    # Donchian — exclude current bar (fix M1)
   128→    ind["donchian_upper"] = np.zeros(n)
   129→    ind["donchian_lower"] = np.zeros(n)
   130→    for i in range(20, n):
   131→        ind["donchian_upper"][i] = np.max(highs[i-20:i])
   132→        ind["donchian_lower"][i] = np.min(lows[i-20:i])
   133→    
   134→    # VWAP (20-period)
   135→    typical = (highs + lows + closes) / 3
   136→    pv = typical * volumes
   137→    ind["vwap"] = rolling_mean(pv, 20) / (rolling_mean(volumes.astype(float), 20) + 1e-10)
   138→    
   139→    # ROC (10)
   140→    ind["roc"] = np.zeros(n)
   141→    if n > 10:
   142→        ind["roc"][10:] = (closes[10:] - closes[:-10]) / (closes[:-10] + 1e-10) * 100
   143→    
   144→    # Stochastic %K (14)
   145→    ind["stoch_k"] = np.full(n, 50.0)
   146→    for i in range(14, n):
   147→        hh = np.max(highs[i-14:i+1])
   148→        ll = np.min(lows[i-14:i+1])
   149→        if hh > ll:
   150→            ind["stoch_k"][i] = (closes[i] - ll) / (hh - ll) * 100
   151→    
   152→    # ADX — proper Wilder formula (fix C10)
   153→    if n >= 28:
   154→        up_move = highs - np.roll(highs, 1)
   155→        down_move = np.roll(lows, 1) - lows
   156→        plus_dm = np.where((up_move > down_move) & (up_move > 0), up_move, 0.0)
   157→        minus_dm = np.where((down_move > up_move) & (down_move > 0), down_move, 0.0)
   158→        
   159→        atr_adx = wilder_smooth(tr, 14)
   160→        plus_di = 100 * wilder_smooth(plus_dm, 14) / (atr_adx + 1e-10)
   161→        minus_di = 100 * wilder_smooth(minus_dm, 14) / (atr_adx + 1e-10)
   162→        dx = 100 * np.abs(plus_di - minus_di) / (plus_di + minus_di + 1e-10)
   163→        ind["adx"] = wilder_smooth(dx, 14)
   164→    else:
   165→        ind["adx"] = np.zeros(n)
   166→    
   167→    # MACD
   168→    ema12 = ema(closes, 12)
   169→    ema26 = ema(closes, 26)
   170→    macd_line = ema12 - ema26
   171→    macd_signal = ema(macd_line, 9)
   172→    ind["macd_hist"] = macd_line - macd_signal
   173→    ind["macd_line"] = macd_line
   174→    ind["macd_signal"] = macd_signal
   175→    
   176→    # 3-candle patterns — proper warmup (fix M2)
   177→    ind["all_up"] = np.zeros(n)
   178→    ind["all_down"] = np.zeros(n)
   179→    for i in range(2, n):
   180→        ind["all_up"][i] = 1.0 if (closes[i] > closes[i-1] > closes[i-2]) else 0.0
   181→        ind["all_down"][i] = 1.0 if (closes[i] < closes[i-1] < closes[i-2]) else 0.0
   182→    
   183→    # Z-score
   184→    ind["zscore"] = (closes - sma20) / (std20 + 1e-10)
   185→    
   186→    # Supertrend — proper recursive (fix C12)
   187→    if n >= 12:
   188→        st_factor = 3.0
   189→        st_upper = np.zeros(n)
   190→        st_lower = np.zeros(n)
   191→        st_dir = np.ones(n)  # 1=up, -1=down
   192→        
   193→        hl2 = (highs + lows) / 2
   194→        st_upper[0] = hl2[0] + st_factor * ind["atr"][0]
   195→        st_lower[0] = hl2[0] - st_factor * ind["atr"][0]
   196→        
   197→        for i in range(1, n):
   198→            # Upper band
   199→            new_upper = hl2[i] + st_factor * ind["atr"][i]
   200→            st_upper[i] = new_upper if (new_upper < st_upper[i-1] or closes[i-1] > st_upper[i-1]) else st_upper[i-1]
   201→            # Lower band
   202→            new_lower = hl2[i] - st_factor * ind["atr"][i]
   203→            st_lower[i] = new_lower if (new_lower > st_lower[i-1] or closes[i-1] < st_lower[i-1]) else st_lower[i-1]
   204→            # Direction
   205→            if st_dir[i-1] == 1:
   206→                st_dir[i] = -1 if closes[i] < st_lower[i] else 1
   207→            else:
   208→                st_dir[i] = 1 if closes[i] > st_upper[i] else -1
   209→        
   210→        ind["st_flip_up"] = ((np.roll(st_dir, 1) == -1) & (st_dir == 1)).astype(float)
   211→        ind["st_flip_dn"] = ((np.roll(st_dir, 1) == 1) & (st_dir == -1)).astype(float)
   212→        ind["st_flip_up"][:1] = 0
   213→        ind["st_flip_dn"][:1] = 0
   214→    else:
   215→        ind["st_flip_up"] = np.zeros(n)
   216→        ind["st_flip_dn"] = np.zeros(n)
   217→    
   218→    # Heikin-Ashi — correct recursive (fix C11)
   219→    ha_close = np.zeros(n)
   220→    ha_open = np.zeros(n)
   221→    ha_close[0] = (opens[0] + highs[0] + lows[0] + closes[0]) / 4
   222→    ha_open[0] = (opens[0] + closes[0]) / 2
   223→    for i in range(1, n):
   224→        ha_close[i] = (opens[i] + highs[i] + lows[i] + closes[i]) / 4
   225→        ha_open[i] = (ha_open[i-1] + ha_close[i-1]) / 2
   226→    
   227→    ind["ha_green"] = (ha_close > ha_open).astype(float)
   228→    ind["ha_red"] = (ha_close < ha_open).astype(float)
   229→    ind["ha_close"] = ha_close
   230→    
   231→    # Golden / Death cross (SMA50 / SMA200)
   232→    if n >= 200:
   233→        sma200 = rolling_mean(closes, 200)
   234→        ind["golden_cross"] = ((np.roll(ind["sma50"], 1) <= np.roll(sma200, 1)) &
   235→                                (ind["sma50"] > sma200)).astype(float)
   236→        ind["death_cross"] = ((np.roll(ind["sma50"], 1) >= np.roll(sma200, 1)) &
   237→                               (ind["sma50"] < sma200)).astype(float)
   238→    else:
   239→        ind["golden_cross"] = np.zeros(n)
   240→        ind["death_cross"] = np.zeros(n)
   241→    
   242→    # BB squeeze (bandwidth at 125-candle low)
   243→    ind["is_squeeze"] = np.zeros(n)
   244→    if n >= 125:
   245→        for i in range(125, n):
   246→            min_bw = np.min(ind["bb_width"][i-125:i])
   247→            ind["is_squeeze"][i] = 1.0 if ind["bb_width"][i] <= min_bw * 1.1 else 0.0
   248→    
   249→    # Dual Thrust (5-candle)
   250→    ind["dt_upper"] = np.zeros(n)
   251→    ind["dt_lower"] = np.zeros(n)
   252→    for i in range(5, n):
   253→        hh = np.max(highs[i-5:i])
   254→        lc = np.min(closes[i-5:i])
   255→        hc = np.max(closes[i-5:i])
   256→        ll = np.min(lows[i-5:i])
   257→        dt_range = max(hh - lc, hc - ll)
   258→        ind["dt_upper"][i] = opens[i] + 0.5 * dt_range
   259→        ind["dt_lower"][i] = opens[i] - 0.5 * dt_range
   260→    
   261→    # Awesome Oscillator cross
   262→    if n >= 34:
   263→        medians = (highs + lows) / 2
   264→        ao = rolling_mean(medians, 5) - rolling_mean(medians, 34)
   265→        prev_ao = np.zeros(n)
   266→        prev_ao[1:] = ao[:-1]
   267→        ind["ao_cross_up"] = ((prev_ao < 0) & (ao > 0)).astype(float)
   268→        ind["ao_cross_dn"] = ((prev_ao > 0) & (ao < 0)).astype(float)
   269→    else:
   270→        ind["ao_cross_up"] = np.zeros(n)
   271→        ind["ao_cross_dn"] = np.zeros(n)
   272→    
   273→    # Opening Range (first candle of day)
   274→    ind["or_high"] = np.zeros(n)
   275→    ind["or_low"] = np.zeros(n)
   276→    # Simplified: use first candle of each 78-candle block (1 trading day)
   277→    for i in range(n):
   278→        day_start = (i // 78) * 78
   279→        if day_start < n:
   280→            ind["or_high"][i] = highs[day_start]
   281→            ind["or_low"][i] = lows[day_start]
   282→    
   283→    # Fix NaN
   284→    for key in ind:
   285→        ind[key] = np.nan_to_num(ind[key], nan=0.0, posinf=0.0, neginf=0.0)
   286→    
   287→    return ind

   290→# ─── Vectorized backtest with Numba ─────────────────────────────────────────
   291→
   292→try:
   293→    from numba import njit
   294→    HAS_NUMBA = True
   295→except ImportError:
   296→    HAS_NUMBA = False
   297→    def njit(*args, **kwargs):
   298→        def decorator(func):
   299→            return func
   300→        if len(args) == 1 and callable(args[0]):
   301→            return args[0]
   302→        return decorator

   304→
   305→@njit(cache=True)
   306→def _backtest_numba(entry_long, entry_short, exit_long, exit_short,
   307→                    closes, highs, lows, tp_pct, hold_ticks_candles,
   308→                    commission, position_size):
   309→    """Numba-compiled backtest loop. 100-1000x faster than pure Python."""
   310→    n = len(closes)
   311→    position = np.zeros(n)
   312→    balance = 10000.0
   313→    entry_price = 0.0
   314→    entry_bar = 0
   315→    trades = 0
   316→    wins = 0
   317→    
   318→    holding = 0  # 0=flat, 1=long, -1=short
   319→    
   320→    for i in range(50, n):
   321→        # === EXIT CHECK (only if holding and past entry bar) ===
   322→        if holding != 0 and i > entry_bar:
   323→            should_exit = False
   324→            
   325→            if holding == 1 and exit_long[i]:
   326→                should_exit = True
   327→            elif holding == -1 and exit_short[i]:
   328→                should_exit = True
   329→            
   330→            # Hold time exit
   331→            if i - entry_bar >= hold_ticks_candles:
   332→                should_exit = True
   333→            
   334→            # Take-profit
   335→            if holding == 1 and tp_pct > 0:
   336→                profit = (highs[i] - entry_price) / entry_price
   337→                if profit >= tp_pct:
   338→                    should_exit = True
   339→            elif holding == -1 and tp_pct > 0:
   340→                profit = (entry_price - lows[i]) / entry_price
   341→                if profit >= tp_pct:
   342→                    should_exit = True
   343→            
   344→            if should_exit:
   345→                # Close position
   346→                if holding == 1:
   347→                    pnl = (closes[i] - entry_price) * (balance * position_size / entry_price)
   348→                else:
   349→                    pnl = (entry_price - closes[i]) * (balance * position_size / entry_price)
   350→                pnl -= balance * position_size * commission  # exit commission
   351→                balance += pnl
   352→                trades += 1
   353→                if pnl > 0:
   354→                    wins += 1
   355→                holding = 0
   356→        
   357→        # === ENTRY CHECK ===
   358→        if holding == 0:
   359→            if entry_long[i]:
   360→                holding = 1
   361→                entry_price = closes[i]
   362→                entry_bar = i
   363→                balance -= balance * position_size * commission  # entry commission
   364→            elif entry_short[i]:
   365→                holding = -1
   366→                entry_price = closes[i]
   367→                entry_bar = i
   368→                balance -= balance * position_size * commission  # entry commission
   369→        
   370→        position[i] = holding
   371→    
   372→    # Close any remaining position at last close
   373→    if holding != 0:
   374→        i = n - 1
   375→        if holding == 1:
   376→            pnl = (closes[i] - entry_price) * (balance * position_size / entry_price)
   377→        else:
   378→            pnl = (entry_price - closes[i]) * (balance * position_size / entry_price)
   379→        pnl -= balance * position_size * commission
   380→        balance += pnl
   381→        trades += 1
   382→        if pnl > 0:
   383→            wins += 1
   384→    
   385→    return balance - 10000.0, trades, wins, position

   388→def vectorized_backtest(ind: Dict[str, np.ndarray], closes: np.ndarray,
   389→                        highs: np.ndarray, lows: np.ndarray,
   390→                        strategy: str, params: dict,
   391→                        commission: float = 0.0005) -> dict:
   392→    """Evaluate one strategy+params combination."""
   393→    n = len(closes)
   394→    
   395→    entry_long = np.zeros(n, dtype=np.float64)
   396→    entry_short = np.zeros(n, dtype=np.float64)
   397→    exit_long = np.zeros(n, dtype=np.float64)
   398→    exit_short = np.zeros(n, dtype=np.float64)
   399→    
   400→    sma_mult = params.get("entry_sma_mult", 0.999)
   401→    rsi_min = params.get("entry_rsi_min", 30)
   402→    rsi_max = params.get("entry_rsi_max", 55)
   403→    tp_pct = params.get("take_profit_pct", 0.0)
   404→    hold_ticks = params.get("hold_ticks", 60)
   405→    exit_sma = params.get("exit_sma_mult", 1.003)
   406→    
   407→    # FIX C2: hold_ticks is in 10s ticks, 1 candle = 5min = 30 ticks
   408→    hold_ticks_candles = max(1, hold_ticks // 30)
   409→    
   410→    sma5 = ind["sma5"]
   411→    sma14 = ind["sma14"]
   412→    sma20 = ind["sma20"]
   413→    sma50 = ind["sma50"]
   414→    rsi = ind["rsi14"]
   415→    rsi2 = ind["rsi2"]
   416→    close = closes
   417→    
   418→    # === ALL 22 STRATEGIES (fix M12) ===
   419→    
   420→    if strategy == "multi_timeframe":
   421→        entry_short[:] = ((sma5 < sma14 * sma_mult) & (rsi >= rsi_min) & (rsi <= rsi_max) &
   422→                          (ind["all_down"] > 0) & (sma20 < sma14))
   423→        entry_long[:] = ((sma5 > sma14 * (2 - sma_mult)) & (rsi >= 40) & (rsi <= 60) &
   424→                         (ind["all_up"] > 0) & (sma20 > sma14))
   425→        exit_long[:] = sma5 < sma14
   426→        exit_short[:] = sma5 > sma14
   427→    
   428→    elif strategy == "v2_short":
   429→        entry_short[:] = ((sma5 < sma14 * sma_mult) & (rsi >= rsi_min) & (rsi <= rsi_max) &
   430→                          (ind["all_down"] > 0))
   431→        entry_long[:] = ((sma5 > sma14 * (2 - sma_mult)) & (rsi >= 25) & (rsi <= 40) &
   432→                         (ind["all_up"] > 0))
   433→        exit_long[:] = (sma5 < sma14 * (2 - exit_sma)) & (rsi < 35)
   434→        exit_short[:] = (sma5 > sma14 * exit_sma) & (rsi > 65)
   435→    
   436→    elif strategy == "v2_inverted":
   437→        entry_long[:] = ((sma5 < sma14 * sma_mult) & (rsi >= 20) & (rsi <= 40) &
   438→                         (ind["all_down"] > 0))
   439→        entry_short[:] = ((sma5 > sma14 * (2 - sma_mult)) & (rsi >= 60) & (rsi <= 80) &
   440→                          (ind["all_up"] > 0))
   441→        exit_long[:] = rsi > 60
   442→        exit_short[:] = rsi < 40
   443→    
   444→    elif strategy == "mean_reversion":
   445→        entry_long[:] = rsi < 25
   446→        entry_short[:] = rsi > 75
   447→        exit_long[:] = rsi > 50
   448→        exit_short[:] = rsi < 50
   449→    
   450→    elif strategy == "trend_follow":
   451→        # FIX C9: proper trend-following entries
   452→        entry_long[:] = (sma5 > sma14 * (2 - sma_mult)) & (rsi > 50) & (ind["all_up"] > 0)
   453→        entry_short[:] = (sma5 < sma14 * sma_mult) & (rsi < 50) & (ind["all_down"] > 0)
   454→        exit_long[:] = sma5 < sma14
   455→        exit_short[:] = sma5 > sma14
   456→    
   457→    elif strategy == "random_hold_short":
   458→        entry_short[:] = ((sma5 < sma14 * sma_mult) & (rsi >= rsi_min) & (rsi <= rsi_max) &
   459→                          (ind["all_down"] > 0))

   638→    position_lagged = np.zeros_like(position)
   639→    position_lagged[1:] = position[:-1]  # shift: position[i-1]

   645→    
   646→    # Max drawdown (fix M8: multiplicative)
   647→    equity = 10000 * np.cumprod(1 + strategy_returns * position_size)
   648→    peak = np.maximum.accumulate(equity)
   649→    dd = (peak - equity) / (peak + 1e-10)
   650→    max_dd = np.max(dd) if len(dd) > 0 else 0
   651→

   656→        "win_rate": float(win_rate),

   658→        "max_dd": float(max_dd),
