   130→   130→/**
   131→   131→ * Simple causal rolling mean via cumsum.
   132→   132→ *   result[i < w]  = cumsum[i] / w   (matches numpy rolling_mean with cumsum trick:
   133→   133→ *                                   warmup gets partial sum / w, NOT / (i+1))
   134→   134→ *   result[i >= w] = (cumsum[i] - cumsum[i-w]) / w
   135→   135→ *
   136→   136→ * Identical to the `rolling_mean()` helper in ml_features.py.
   137→   137→ */
   138→   138→function rollingMean(arr: number[], w: number): number[] {
   139→   139→  const n = arr.length
   140→   140→  const result = new Array<number>(n).fill(0)
   141→   141→  if (n === 0) return result
   142→   142→  const cumsum = new Array<number>(n)
   143→   143→  let s = 0
   144→   144→  for (let i = 0; i < n; i++) {
   145→   145→    s += arr[i]
   146→   146→    cumsum[i] = s
   147→   147→  }
   148→   148→  for (let i = 0; i < n; i++) {
   149→   149→    if (i < w) {
   150→   150→      // Python rolling_mean returns cumsum[i] / w (smaller than true mean during warmup).
   151→   151→      result[i] = cumsum[i] / w
   152→   152→    } else {
   153→   153→      result[i] = (cumsum[i] - cumsum[i - w]) / w
   154→   154→    }
   155→   155→  }
   156→   156→  return result
   157→   157→}
   158→   158→
   159→   159→/**
   160→   160→ * Causal SMA used in ml_features.py `causal_sma()` — DIFFERENT from rolling_mean:
   161→   161→ *   result[i < w]  = cumsum[i] / (i + 1)   (true partial mean)
   162→   162→ *   result[i >= w] = (cumsum[i] - cumsum[i-w]) / w
   163→   163→ *
   164→   164→ * Used for sma5/14/20/50.
   165→   165→ */
   166→   166→function causalSMA(arr: number[], w: number): number[] {
   167→   167→  const n = arr.length
   168→   168→  const result = new Array<number>(n).fill(0)
   169→   169→  if (n === 0) return result
   170→   170→  const cumsum = new Array<number>(n)
   171→   171→  let s = 0
   172→   172→  for (let i = 0; i < n; i++) {
   173→   173→    s += arr[i]
   174→   174→    cumsum[i] = s
   175→   175→  }
   176→   176→  for (let i = 0; i < n; i++) {
   177→   177→    if (i < w) {
   178→   178→      result[i] = cumsum[i] / (i + 1)
   179→   179→    } else {
   180→   180→      result[i] = (cumsum[i] - cumsum[i - w]) / w
   181→   181→    }
   182→   182→  }
   183→   183→  return result
   184→   184→}

   186→   186→/** Standard EMA with smoothing factor k = 2/(period+1). Matches `ema()` in ml_features.py. */
   187→   187→function ema(arr: number[], period: number): number[] {
   188→   188→  const n = arr.length
   189→   189→  const result = new Array<number>(n).fill(0)
   190→   190→  if (n === 0) return result
   191→   191→  const k = 2 / (period + 1)
   192→   192→  result[0] = arr[0]
   193→   193→  for (let i = 1; i < n; i++) {
   194→   194→    result[i] = arr[i] * k + result[i - 1] * (1 - k)
   195→   195→  }
   196→   196→  return result
   197→   197→}

   200→   200→ * Compute the 31-feature vector at the LAST bar of `candles`.
   201→   201→ *
   202→   202→ * Returns a fixed-length array of 31 numbers, in alphabetical feature order.

   231→   231→    close[i] = candles[i].close
   232→   232→    high[i] = candles[i].high
   233→   233→    low[i] = candles[i].low
   234→   234→    vol[i] = candles[i].volume
   235→   235→    time[i] = candles[i].time
   236→   236→  }

   238→   238→
   239→   239→  // ── Returns (causal, no wraparound) ──────────────────────────────────────────
   240→   240→  const ret1 = new Array<number>(n).fill(0)
   241→   241→  const ret5 = new Array<number>(n).fill(0)
   242→   242→  const ret10 = new Array<number>(n).fill(0)
   243→   243→  const ret30 = new Array<number>(n).fill(0)
   244→   244→  const ret5log = new Array<number>(n).fill(0)
   245→   245→  for (let i = 1; i < n; i++) {

   247→   247→  }

   251→   251→  }
   252→   252→  for (let i = 10; i < n; i++) {
   253→   253→    ret10[i] = (close[i] - close[i - 10]) / (close[i - 10] + EPS)
   254→   254→  }
   255→   255→  for (let i = 30; i < n; i++) {
   256→   256→    ret30[i] = (close[i] - close[i - 30]) / (close[i - 30] + EPS)
   257→   257→  }
   258→   258→
   259→   259→  // ── SMAs ───────────────────────────────────────────────────────────────────
   260→   260→  const sma5 = causalSMA(close, 5)
   261→   261→  const sma14 = causalSMA(close, 14)
   262→   262→  const sma20 = causalSMA(close, 20)
   263→   263→  const sma50 = causalSMA(close, 50)
   264→   264→  // Approx SMAs for higher-TF features
   265→   265→  const sma120 = causalSMA(close, 120)  // ~10 hours of 5min candles
   266→   266→  const sma144 = causalSMA(close, 144)  // ~1 trading day of 5min candles
   267→   267→
   268→   268→  const sma5_sma14 = sma5.map((v, i) => v / (sma14[i] + EPS))
   269→   269→  const sma14_sma20 = sma14.map((v, i) => v / (sma20[i] + EPS))
   270→   270→  const sma20_sma50 = sma20.map((v, i) => v / (sma50[i] + EPS))
   271→   271→  const price_sma20 = close.map((v, i) => v / (sma20[i] + EPS))
   272→   272→  const price_sma50 = close.map((v, i) => v / (sma50[i] + EPS))
   273→   273→
   274→   274→  // ── RSI (SIMPLE rolling mean of gains/losses, NOT Wilder) ─────────────────────
   275→   275→  const deltas = new Array<number>(n)
   276→   276→  deltas[0] = 0
   277→   277→  for (let i = 1; i < n; i++) {
   278→   278→    deltas[i] = close[i] - close[i - 1]
   279→   279→  }
   280→   280→  const gains = new Array<number>(n)
   281→   281→  const losses = new Array<number>(n)
   282→   282→  for (let i = 0; i < n; i++) {

   301→   301→    rsi2[i] = 100 - 100 / (1 + rs)
   302→   302→  }
   303→   303→
   304→   304→  // ── Bollinger Bands ─────────────────────────────────────────────────────────
   305→   305→  // Python: std20[i] = np.std(close5[max(0,i-19):i+1], ddof=1) if i>=19 else 0
   306→   306→  //       = sample stddev over last 20 closes (inclusive of i)
   307→   307→  const std20 = new Array<number>(n).fill(0)
   308→   308→  for (let i = 19; i < n; i++) {
   309→   309→    let mean = 0
   310→   310→    for (let j = i - 19; j <= i; j++) mean += close[j]
   311→   311→    mean /= 20
   312→   312→    let sq = 0
   313→   313→    for (let j = i - 19; j <= i; j++) {

   334→   334→  const ema12 = ema(close, 12)

   336→   336→  const macdLine = new Array<number>(n)
   337→   337→  for (let i = 0; i < n; i++) macdLine[i] = ema12[i] - ema26[i]
   338→   338→  const macdSignal = ema(macdLine, 9)
   339→   339→  const macdHist = new Array<number>(n)
   340→   340→  for (let i = 0; i < n; i++) macdHist[i] = macdLine[i] - macdSignal[i]
   341→   341→  // Normalize by close (matches ml_features.py)
   342→   342→  const macd_line = macdLine.map((v, i) => v / (close[i] + EPS))
   343→   343→  const macd_signal = macdSignal.map((v, i) => v / (close[i] + EPS))
   344→   344→
   345→   345→  // ── ATR (SIMPLE rolling mean of TR, NOT Wilder) ──────────────────────────────
   346→   346→  // Python: tr = max(h-l, |h-prev_close|, |l-prev_close|) with np.roll(close, 1).
   347→   347→  // np.roll wraps (last→first), but first bar is in warmup. We use causal
   348→   348→  // prev_close (close[i-1] for i>0, close[0] for i=0) which matches Python
   349→   349→  // for i>=1 exactly.
   350→   350→  const tr = new Array<number>(n)
   351→   351→  for (let i = 0; i < n; i++) {
   352→   352→    const prevClose = i === 0 ? close[0] : close[i - 1]
   353→   353→    const a = high[i] - low[i]
   354→   354→    const b = Math.abs(high[i] - prevClose)
   355→   355→    const c = Math.abs(low[i] - prevClose)
   356→   356→    tr[i] = Math.max(a, Math.max(b, c))
   357→   357→  }

   359→   359→  const atr_pct = atr14.map((v, i) => v / (close[i] + EPS))

   362→   362→  const volAvg20 = rollingMean(vol, 20)
   363→   363→  const vol_ratio = vol.map((v, i) => v / (volAvg20[i] + EPS))
   364→   364→
   365→   365→  // ── OBV slope (10-bar window, NOT 5) ─────────────────────────────────────────
   366→   366→  const obv = new Array<number>(n)
   367→   367→  obv[0] = 0
   368→   368→  for (let i = 1; i < n; i++) {
   369→   369→    obv[i] = obv[i - 1] + (deltas[i] > 0 ? vol[i] : -vol[i])
   370→   370→  }
   371→   371→  const obv_slope = new Array<number>(n).fill(0)
   372→   372→  for (let i = 10; i < n; i++) {
   373→   373→    obv_slope[i] = (obv[i] - obv[i - 10]) / (obv[i - 10] + EPS)
   374→   374→  }
   375→   375→
   376→   376→  // ── Stochastic %K (14-bar high/low, INCLUSIVE of current bar) ────────────────
   377→   377→  // Python: hh14[i] = max(high[max(0,i-13):i+1]); ll14[i] = min(low[...])
   378→   378→  const stoch_k = new Array<number>(n)
   379→   379→  for (let i = 0; i < n; i++) {
   380→   380→    const start = Math.max(0, i - 13)
   381→   381→    let hh = -Infinity
   382→   382→    let ll = Infinity
   383→   383→    for (let j = start; j <= i; j++) {
   384→   384→      if (high[j] > hh) hh = high[j]
   385→   385→      if (low[j] < ll) ll = low[j]
   386→   386→    }
   387→   387→    stoch_k[i] = ((close[i] - ll) / (hh - ll + EPS)) * 100
   388→   388→  }
   389→   389→
   390→   390→  // ── ADX (SIMPLIFIED — NOT Wilder) ───────────────────────────────────────────
   391→   391→  // Python:
   392→   392→  //   up_moves   = where(deltas > 0, 1, 0)
   393→   393→  //   down_moves = where(deltas < 0, 1, 0)
   394→   394→  //   adx = abs(rolling_mean(up_moves, 14) - rolling_mean(down_moves, 14)) * 100
   395→   395→  const up_moves = new Array<number>(n)
   396→   396→  const down_moves = new Array<number>(n)
   397→   397→  for (let i = 0; i < n; i++) {
   398→   398→    up_moves[i] = deltas[i] > 0 ? 1 : 0
   399→   399→    down_moves[i] = deltas[i] < 0 ? 1 : 0
   400→   400→  }
   401→   401→  const rm_up = rollingMean(up_moves, 14)
   402→   402→  const rm_dn = rollingMean(down_moves, 14)
   403→   403→  const adx = new Array<number>(n)
   404→   404→  for (let i = 0; i < n; i++) {
   405→   405→    adx[i] = Math.abs(rm_up[i] - rm_dn[i]) * 100
   406→   406→  }
   407→   407→
   408→   408→  // ── Higher timeframe context (approximations from 5min candles) ──────────────
   409→   409→  // ml_features.py uses real 1hour and 1day candle series from the multi-TF
   410→   410→  // data pipeline.  The trader server only has 5min candles, so we approximate.
   411→   411→  // This matches the existing v2 meta_selector.ts approach (tested & deployed).
   412→   412→  const ret1h = idx >= 12 ? (close[idx] - close[idx - 12]) / (close[idx - 12] + EPS) : 0
   413→   413→  const ret1d = idx >= 144 ? (close[idx] - close[idx - 144]) / (close[idx - 144] + EPS) : 0
   414→   414→  // 1h_trend: close / sma10(close1h)  ≈ close / sma120(close5min)  [10 hours]
   415→   415→  const trend1h = close[idx] / (sma120[idx] + EPS)
   416→   416→  // 1d_trend: close / sma5(close1d)   ≈ close / sma144(close5min)  [~1 trading day]
   417→   417→  const trend1d = close[idx] / (sma144[idx] + EPS)
   418→   418→  // 1h_rsi: approximation — use 5min RSI14 (no real 1hour close series available)
   419→   419→  const rsi1h = rsi14[idx]
   420→   420→
   421→   421→  // ── Time features (MSK timezone, UTC+3) ──────────────────────────────────────
   422→   422→  // ml_features.py:
   423→   423→  //   ts_seconds = time5 / 1000                    # time5 is in ms
   424→   424→  //   hours = (ts_seconds // 3600 + 3) % 24        # MSK = UTC+3
   425→   425→  //   dow   = (ts_seconds // 86400 + 4) % 7        # Sun=0...Sat=6 (matches JS getUTCDay)
   426→   426→  //   hour         = hours / 24.0
   427→   427→  //   day_of_week  = dow / 7.0
   428→   428→  //
   429→   429→  // regime_detector.ts Candle.time is "unix seconds (or any monotonic unit)".
   430→   430→  // The trader server's Candle.time is in MILLISECONDS (existing meta_selector.ts
   431→   431→  // does `new Date(ts)` directly, which expects ms).  We auto-detect: if ts > 1e12
   432→   432→  // (year 2001+), treat as ms; else seconds.
   433→   433→  const ts = time[idx]
   434→   434→  const tsMs = ts > 1e12 ? ts : ts * 1000
   435→   435→  const date = new Date(tsMs)
   436→   436→  const hour = ((date.getUTCHours() + 3) % 24) / 24.0
   437→   437→  const day_of_week = date.getUTCDay() / 7.0
   438→   438→
   439→   439→  // ── Assemble feature vector in ALPHABETICAL order ────────────────────────────
   440→   440→  // (matches training metadata feature_names exactly)
   441→   441→  const features: Record<string, number> = {
   442→   442→    '1d_ret': ret1d,
   443→   443→    '1d_trend': trend1d,
   444→   444→    '1h_ret': ret1h,
   445→   445→    '1h_rsi': rsi1h,
   446→   446→    '1h_trend': trend1h,
   447→   447→    'adx': adx[idx],
   448→   448→    'atr_pct': atr_pct[idx],
   449→   449→    'bb_pct_b': bbPctB[idx],
   450→   450→    'bb_width': bbWidth[idx],
   451→   451→    'day_of_week': day_of_week,
   452→   452→    'hour': hour,
   453→   453→    'macd_hist': macdHist[idx],
   454→   454→    'macd_line': macd_line[idx],
   455→   455→    'macd_signal': macd_signal[idx],
   456→   456→    'obv_slope': obv_slope[idx],
   457→   457→    'price_bb_lower': price_bb_lower[idx],
   458→   458→    'price_bb_upper': price_bb_upper[idx],
   459→   459→    'price_sma20': price_sma20[idx],
   460→   460→    'price_sma50': price_sma50[idx],
   461→   461→    'ret_1': ret1[idx],
   462→   462→    'ret_10': ret10[idx],
   463→   463→    'ret_30': ret30[idx],
   464→   464→    'ret_5': ret5[idx],
   465→   465→    'ret_5_log': ret5log[idx],
   466→   466→    'rsi14': rsi14[idx],
   467→   467→    'rsi2': rsi2[idx],
   468→   468→    'sma14_sma20': sma14_sma20[idx],
   469→   469→    'sma20_sma50': sma20_sma50[idx],
   470→   470→    'sma5_sma14': sma5_sma14[idx],
   471→   471→    'stoch_k': stoch_k[idx],
   472→   472→    'vol_ratio': vol_ratio[idx],
   473→   473→  }
   474→   474→
   475→   475→  // Build the vector in the exact order specified by the task / training metadata.
   476→   476→  const FEATURE_NAMES = [
   477→   477→    '1d_ret', '1d_trend', '1h_ret', '1h_rsi', '1h_trend', 'adx', 'atr_pct',
   478→   478→    'bb_pct_b', 'bb_width', 'day_of_week', 'hour', 'macd_hist', 'macd_line',
   479→   479→    'macd_signal', 'obv_slope', 'price_bb_lower', 'price_bb_upper',
   480→   480→    'price_sma20', 'price_sma50', 'ret_1', 'ret_10', 'ret_30', 'ret_5',
   481→   481→    'ret_5_log', 'rsi14', 'rsi2', 'sma14_sma20', 'sma20_sma50', 'sma5_sma14',
   482→   482→    'stoch_k', 'vol_ratio',
   483→   483→  ]
   484→   484→  const out = new Array<number>(FEATURE_NAMES.length)
   485→   485→  for (let i = 0; i < FEATURE_NAMES.length; i++) {
   486→   486→    let v = features[FEATURE_NAMES[i]]
   487→   487→    if (!Number.isFinite(v)) v = 0
   488→   488→    // Python: np.clip(X, -10, 10)
   489→   489→    if (v > 10) v = 10
   490→   490→    else if (v < -10) v = -10
   491→   491→    out[i] = v
   492→   492→  }
   493→   493→  return out
   494→   494→}

   496→   496→// ─── Model loader (with path search + caching) ───────────────────────────────
   497→   497→
   498→   498→/**
   499→   499→ * Resolve the absolute path of a regime model file.
   500→   500→ *
   501→   501→ * Tries, in order:
   502→   502→ *   /opt/ai-trader/data/regime_<name>.json
   503→   503→ *   /opt/ai-trader/src/strategies/regime_<name>.json
   504→   504→ *   <__dirname>/regime_<name>.json
   505→   505→ *   <cwd>/regime_<name>.json
   506→   506→ *
   507→   507→ * Returns the first existing path, or null if none found.
   508→   508→ */
   509→   509→function resolveModelPath(modelFile: string): string | null {
   510→   510→  for (const dir of MODEL_SEARCH_PATHS) {
   511→   511→    const p = path.join(dir, modelFile)
   512→   512→    try {
   513→   513→      if (fs.existsSync(p)) return p
   514→   514→    } catch {
   515→   515→      // ignore — try next
   516→   516→    }
   517→   517→  }
   518→   518→  return null
   519→   519→}
   520→   520→
   521→   521→/** Per-regime model cache: regimeIndex → loaded model (loaded lazily on first hit). */
   522→   522→const _modelByRegime = new Map<number, ReturnType<typeof loadModel> | null>()
   523→   523→
   524→   524→/**
   525→   525→ * Get the (cached) ML model for a regime.
   526→   526→ *
   527→   527→ * Returns null if:
   528→   528→ *   - The regime has no model file (OVERSOLD_BOUNCE, OVERBOUGHT_REVERSAL)
   529→   529→ *   - The model file cannot be found in any search path
   530→   530→ *
   531→   531→ * On first call for a regime, the model is loaded from disk and cached.
   532→   532→ * Subsequent calls return the cached instance without disk I/O.
   533→   533→ */
   534→   534→function getRegimeModel(regimeIdx: number): ReturnType<typeof loadModel> | null {
   535→   535→  if (_modelByRegime.has(regimeIdx)) {
   536→   536→    return _modelByRegime.get(regimeIdx) ?? null
   537→   537→  }
   538→   538→  const modelFile = REGIME_MODEL_FILES[regimeIdx]
   539→   539→  if (!modelFile) {
   540→   540→    _modelByRegime.set(regimeIdx, null)
   541→   541→    return null
   542→   542→  }
   543→   543→  const modelPath = resolveModelPath(modelFile)
   544→   544→  if (!modelPath) {
   545→   545→    console.warn(
   546→   546→      `[MetaSelectorV4] WARN: model file not found: ${modelFile} ` +
   547→   547→        `(searched: ${MODEL_SEARCH_PATHS.join(', ')})`,
   548→   548→    )
   549→   549→    _modelByRegime.set(regimeIdx, null)
   550→   550→    return null
   551→   551→  }
   552→   552→  try {
   553→   553→    const model = loadModel(modelPath)
   554→   554→    _modelByRegime.set(regimeIdx, model)
   555→   555→    console.log(
   556→   556→      `[MetaSelectorV4] loaded ${modelFile}: nTrees=${model.nTrees} nFeatures=${model.nFeatures}`,
   557→   557→    )
   558→   558→    return model
   559→   559→  } catch (e: any) {
   560→   560→    console.error(`[MetaSelectorV4] ERROR loading ${modelFile}: ${e.message}`)
   561→   561→    _modelByRegime.set(regimeIdx, null)
   562→   562→    return null
   563→   563→  }
   564→   564→}
   565→   565→
   566→   566→/** Clear all cached models (useful for hot-reload / testing). */
   567→   567→export function clearMetaV4ModelCache(): void {
   568→   568→  _modelByRegime.clear()
   569→   569→}

   571→   571→// ─── Strategy class ────────────────────────────────────────────────────────────
   572→   572→
   573→   573→/**
   574→   574→ * MetaSelectorV4Strategy — the v4 ML orchestrator.
   575→   575→ *
   576→   576→ *   detectRegime → loadModel → predict_proba → LONG / SHORT / FLAT
   577→   577→ *
   578→   578→ * Fallback for regimes with no ML model (skipped during training due to
   579→   579→ * insufficient samples):
   580→   580→ *   OVERSOLD_BOUNCE     → LONG  (buy the dip — RSI<25 + close>SMA5)
   581→   581→ *   OVERBOUGHT_REVERSAL → SHORT (sell the top — RSI>75 + close<SMA5)
   582→   582→ */
   583→   583→export class MetaSelectorV4Strategy implements IStrategy {
   584→   584→  name = 'meta_selector_v4'
   585→   585→  description =
   586→   586→    'ML v4 orchestrator: detectRegime → per-regime binary XGBoost (P(up)) → LONG/SHORT/FLAT'
   587→   587→
   588→   588→  /** Last prediction details — exposed for logging/monitoring. */
   589→   589→  lastRegimeIdx: number = -1
   590→   590→  lastRegimeName: string = ''
   591→   591→  lastProba: number = 0.5
   592→   592→  lastDecision: number = 0
   593→   593→  lastDecisionLabel: string = 'FLAT'
   594→   594→  lastModelUsed: string = ''
   595→   595→  lastError: string = ''
   596→   596→
   597→   597→  predict(
   598→   598→    candles: Candle[],
   599→   599→    idx: number,
   600→   600→    _hasPosition: boolean,
   601→   601→    _stepsHeld?: number,
   602→   602→    _ctx?: StrategyContext,
   603→   603→  ): number {
   604→   604→    // Reset transient state
   605→   605→    this.lastError = ''
   606→   606→
   607→   607→    // Need enough candle history.  detectRegime requires >100 bars; features
   608→   608→    // need >50; we require 200 for the higher-TF approximations (144-bar SMA).
   609→   609→    if (!candles || candles.length < MIN_CANDLES) {
   610→   610→      this.lastError = `insufficient_history (need >= ${MIN_CANDLES} candles, got ${candles?.length ?? 0})`
   611→   611→      // During warmup, stay flat.
   612→   612→      this.lastDecision = 0
   613→   613→      this.lastDecisionLabel = 'FLAT'
   614→   614→      return 0
   615→   615→    }
   616→   616→
   617→   617→    // 1. Detect regime at the latest bar
   618→   618→    const regime = detectRegime(candles)
   619→   619→    this.lastRegimeIdx = regime.regime
   620→   620→    this.lastRegimeName = regime.regimeName
   621→   621→
   622→   622→    // 2. If regime has no ML model → use rule-based fallback
   623→   623→    const model = getRegimeModel(regime.regime)
   624→   624→    if (!model) {
   625→   625→      let decision: number
   626→   626→      let label: string
   627→   627→      if (regime.regime === 7 /* OVERSOLD_BOUNCE */) {
   628→   628→        decision = 1 // LONG — buy the dip
   629→   629→        label = 'LONG'
   630→   630→      } else if (regime.regime === 8 /* OVERBOUGHT_REVERSAL */) {
   631→   631→        decision = 2 // SHORT — sell the top
   632→   632→        label = 'SHORT'
   633→   633→      } else {
   634→   634→        // Model file not found for a regime that should have one.  Stay flat
   635→   635→        // and log loudly so the operator notices.
   636→   636→        decision = 0
   637→   637→        label = 'FLAT'
   638→   638→        this.lastError = `model_missing_for_${regime.regimeName}`
   639→   639→      }
   640→   640→      this.lastProba = 0.5
   641→   641→      this.lastDecision = decision
   642→   642→      this.lastDecisionLabel = label
   643→   643→      this.lastModelUsed = '(fallback)'
   644→   644→      console.log(
   645→   645→        `[MetaSelectorV4] regime=${regime.regimeName} (idx=${regime.regime}) ` +
   646→   646→          `P(up)=N/A → ${label} [FALLBACK]`,
   647→   647→      )
   648→   648→      return decision
   649→   649→    }
   650→   650→
   651→   651→    // 3. Has ML model → compute features + predict
   652→   652→    this.lastModelUsed = regime.regimeName
   653→   653→    let features: number[]
   654→   654→    try {
   655→   655→      features = computeMetaV4Features(candles)
   656→   656→    } catch (e: any) {
   657→   657→      this.lastError = `feature_error: ${e.message}`
   658→   658→      this.lastDecision = 0
   659→   659→      this.lastDecisionLabel = 'FLAT'
   660→   660→      console.error(`[MetaSelectorV4] feature computation failed: ${e.message}`)
   661→   661→      return 0
   662→   662→    }
   663→   663→
   664→   664→    let proba: number
   665→   665→    try {
   666→   666→      proba = predict_proba(model, features)
   667→   667→    } catch (e: any) {
   668→   668→      this.lastError = `predict_error: ${e.message}`
   669→   669→      this.lastDecision = 0
   670→   670→      this.lastDecisionLabel = 'FLAT'
   671→   671→      console.error(`[MetaSelectorV4] predict_proba failed: ${e.message}`)
   672→   672→      return 0
   673→   673→    }
   674→   674→
   675→   675→    // Guard against NaN/Inf
   676→   676→    if (!Number.isFinite(proba)) {
   677→   677→      proba = 0.5
   678→   678→    }
   679→   679→    this.lastProba = proba
   680→   680→
   681→   681→    // 4. Decision thresholds (from v4 training metadata)
   682→   682→    let decision: number
   683→   683→    let label: string
   684→   684→    if (proba > LONG_THRESHOLD) {
   685→   685→      decision = 1 // LONG
   686→   686→      label = 'LONG'
   687→   687→    } else if (proba < SHORT_THRESHOLD) {
   688→   688→      decision = 2 // SHORT
   689→   689→      label = 'SHORT'
   690→   690→    } else {
   691→   691→      decision = 0 // FLAT
   692→   692→      label = 'FLAT'
   693→   693→    }
   694→   694→    this.lastDecision = decision
   695→   695→    this.lastDecisionLabel = label
   696→   696→
   697→   697→    // 5. Log every prediction
   698→   698→    console.log(
   699→   699→      `[MetaSelectorV4] regime=${regime.regimeName} (idx=${regime.regime}) ` +
   700→   700→        `P(up)=${proba.toFixed(4)} → ${label}`,
   701→   701→    )
   702→   702→
   703→   703→    return decision
   704→   704→  }
   705→   705→}
   706→   706→

   708→   708→export function createMetaSelectorV4Strategy(): MetaSelectorV4Strategy {
   709→   709→  return new MetaSelectorV4Strategy()
