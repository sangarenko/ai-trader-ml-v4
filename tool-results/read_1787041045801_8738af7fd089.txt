     1→     1→#!/usr/bin/env python3
     2→     2→"""Train 12 binary XGBoost classifiers — one per market regime (ML v4).
     3→     3→
     4→     4→For each of 12 regimes (STRONG_TREND_UP, MILD_TREND_UP, RANGE_TIGHT, RANGE_WIDE,
     5→     5→MILD_TREND_DOWN, STRONG_TREND_DOWN, CRASH, OVERSOLD_BOUNCE,
     6→     6→OVERBOUGHT_REVERSAL, BREAKOUT_UP, BREAKDOWN, HIGH_VOL_REGIME):
     7→     7→
     8→     8→  1. Filter bars to that regime
     9→     9→  2. If < 100 samples: skip (rule-based fallback will handle inference)
    10→    10→  3. Compute binary label: y = 1 if forward_return > 0.001 else 0
    11→    11→     where forward_return = close[t+6] / close[t] - 1 (≈30min ahead)
    12→    12→  4. Train XGBoost binary classifier with strong regularization
    13→    13→     (max_depth=3, reg_lambda=10, min_child_weight=30)
    14→    14→  5. Chronological split: 70% train / 15% val / 15% test (per-ticker)
    15→    15→  6. Save .pkl + .json (via booster.save_raw(raw_format='json'))
    16→    16→
    17→    17→Decision rule for inference:
    18→    18→    if P(up) > 0.6  → go LONG
    19→    19→    if P(up) < 0.4  → go SHORT
    20→    20→    otherwise       → FLAT
    21→    21→
    22→    22→Outputs (in /root/ai-trader-evolution/ml/meta_models_v2/):
    23→    23→  regime_<name>.pkl          — full sklearn XGBClassifier (for backtest)
    24→    24→  regime_<name>.json         — XGBoost native JSON (for TS inference)
    25→    25→  regime_models_v4_metadata.json — per-regime metadata (n_samples, val_precision, val_f1, ...)
    26→    26→
    27→    27→Usage:
    28→    28→  python3 train_regime_models_v4.py --days 180
    29→    29→  python3 train_regime_models_v4.py --days 180 --tickers SBER,GAZP

   104→   104→    """Print + append to log file with MSK timestamp."""

   117→   117→# Label computation
   118→   118→# ---------------------------------------------------------------------------
   119→   119→def compute_binary_label(close5: np.ndarray,
   120→   120→                          horizon: int = DEFAULT_HORIZON,
   121→   121→                          threshold: float = DEFAULT_THRESHOLD) -> np.ndarray:
   122→   122→    """Compute binary label y = 1 if forward_return > threshold else 0.
   123→   123→
   124→   124→    forward_return = (close[t + horizon] - close[t]) / close[t]
   125→   125→
   126→   126→    Last `horizon` bars have no forward close — set to -1 (sentinel for "drop").
   127→   127→
   128→   128→    Args:
   129→   129→        close5: 5-min (actually 10-min in v1 pipeline, but "5min" naming kept) close prices
   130→   130→        horizon: forward window in bars
   131→   131→        threshold: minimum forward return to label as 1 (0.001 = 0.1%)
   132→   132→
   133→   133→    Returns:
   134→   134→        int array of length n; values in {0, 1, -1}. -1 = drop (no future data).
   135→   135→    """
   136→   136→    n = len(close5)
   137→   137→    y = np.full(n, -1, dtype=np.int32)
   138→   138→    if n <= horizon:
   139→   139→        return y
   140→   140→    # Forward close: shift by `horizon`. Last `horizon` bars use placeholder (dropped).
   141→   141→    forward_close = np.empty(n)
   142→   142→    forward_close[:-horizon] = close5[horizon:]
   143→   143→    forward_close[-horizon:] = close5[-1]
   144→   144→    forward_return = (forward_close - close5) / (close5 + 1e-10)
   145→   145→    y = (forward_return > threshold).astype(np.int32)
   146→   146→    y[-horizon:] = -1  # mark as drop
   147→   147→    return y

   151→   151→# Training
   152→   152→# ---------------------------------------------------------------------------
   153→   153→def train_one_regime_model(X_train, y_train, X_val, y_val, X_test, y_test,
   154→   154→                           regime_name: str) -> dict:
   155→   155→    """Train one XGBoost binary classifier for a single regime.
   156→   156→
   157→   157→    Returns dict with 'model' and 'metrics', or None if too few pos/neg samples.
   158→   158→    """
   159→   159→    pos = int(y_train.sum())
   160→   160→    neg = int(len(y_train) - pos)
   161→   161→
   162→   162→    if pos < MIN_POS_NEG or neg < MIN_POS_NEG:
   163→   163→        log(f"  [{regime_name}] SKIP: insufficient pos/neg (pos={pos}, neg={neg})")
   164→   164→        return None
   165→   165→
   166→   166→    # scale_pos_weight handles class imbalance (long-biased market → more "up" labels)
   167→   167→    spw = neg / max(1, pos)
   168→   168→    spw = float(max(SPW_MIN, min(SPW_MAX, spw)))
   169→   169→
   170→   170→    log(f"  [{regime_name}] train: n={len(y_train)}, pos={pos} ({pos*100/len(y_train):.1f}%), "
   171→   171→        f"neg={neg}, scale_pos_weight={spw:.3f}")
   172→   172→    log(f"  [{regime_name}] val:   n={len(y_val)}, pos={int(y_val.sum())} ({y_val.mean()*100:.1f}%)")
   173→   173→    log(f"  [{regime_name}] test:  n={len(y_test)}, pos={int(y_test.sum())} ({y_test.mean()*100:.1f}%)")
   174→   174→
   175→   175→    model = xgb.XGBClassifier(
   176→   176→        **XGB_PARAMS,
   177→   177→        scale_pos_weight=spw,
   178→   178→    )
   179→   179→    model.fit(
   180→   180→        X_train, y_train,
   181→   181→        eval_set=[(X_val, y_val)],
   182→   182→        verbose=False,
   183→   183→    )
   184→   184→
   185→   185→    best_iter = getattr(model, "best_iteration", None)
   186→   186→    if best_iter is None:
   187→   187→        best_iter = XGB_PARAMS["n_estimators"]
   188→   188→
   189→   189→    # --- Validation metrics ---
   190→   190→    y_pred_val = (model.predict_proba(X_val)[:, 1] > 0.5).astype(int)
   191→   191→    val_precision = precision_score(y_val, y_pred_val, zero_division=0)
   192→   192→    val_recall = recall_score(y_val, y_pred_val, zero_division=0)
   193→   193→    val_f1 = f1_score(y_val, y_pred_val, zero_division=0)
   194→   194→    val_acc = accuracy_score(y_val, y_pred_val)
   195→   195→
   196→   196→    # --- Test metrics ---
   197→   197→    y_pred_test = (model.predict_proba(X_test)[:, 1] > 0.5).astype(int)
   198→   198→    test_precision = precision_score(y_test, y_pred_test, zero_division=0)
   199→   199→    test_recall = recall_score(y_test, y_pred_test, zero_division=0)
   200→   200→    test_f1 = f1_score(y_test, y_pred_test, zero_division=0)
   201→   201→    test_acc = accuracy_score(y_test, y_pred_test)
   202→   202→
   203→   203→    # --- Precision @ 0.6 threshold (the actual LONG_THRESHOLD used in inference) ---
   204→   204→    y_proba_test = model.predict_proba(X_test)[:, 1]
   205→   205→    n_high_conf = int((y_proba_test > LONG_THRESHOLD).sum())
   206→   206→    if n_high_conf >= 10:
   207→   207→        prec_at_06 = precision_score(
   208→   208→            y_test[y_proba_test > LONG_THRESHOLD],
   209→   209→            y_pred_test[y_proba_test > LONG_THRESHOLD],
   210→   210→            zero_division=0,
   211→   211→        )
   212→   212→    else:
   213→   213→        prec_at_06 = None
   214→   214→
   215→   215→    log(f"  [{regime_name}] VAL:  precision={val_precision*100:.1f}% recall={val_recall*100:.1f}% "
   216→   216→        f"f1={val_f1*100:.1f}% acc={val_acc*100:.1f}%")
   217→   217→    log(f"  [{regime_name}] TEST: precision={test_precision*100:.1f}% recall={test_recall*100:.1f}% "
   218→   218→        f"f1={test_f1*100:.1f}% acc={test_acc*100:.1f}%")
   219→   219→    if prec_at_06 is not None:
   220→   220→        log(f"  [{regime_name}] TEST precision@P>0.6: {prec_at_06*100:.1f}% (n={n_high_conf})")
   221→   221→    log(f"  [{regime_name}] best_iteration={best_iter}")
   222→   222→
   223→   223→    return {
   224→   224→        "model": model,
   225→   225→        "metrics": {
   226→   226→            "n_train": int(len(y_train)),
   227→   227→            "train_pos": int(pos),
   228→   228→            "train_neg": int(neg),
   229→   229→            "scale_pos_weight": float(spw),
   230→   230→            "best_iteration": int(best_iter),
   231→   231→            # validation
   232→   232→            "val_precision": float(val_precision),
   233→   233→            "val_recall": float(val_recall),
   234→   234→            "val_f1": float(val_f1),
   235→   235→            "val_accuracy": float(val_acc),
   236→   236→            "val_n": int(len(y_val)),
   237→   237→            # test
   238→   238→            "test_precision": float(test_precision),
   239→   239→            "test_recall": float(test_recall),
   240→   240→            "test_f1": float(test_f1),
   241→   241→            "test_accuracy": float(test_acc),
   242→   242→            "test_n": int(len(y_test)),
   243→   243→            # high-confidence precision
   244→   244→            "test_precision_at_0.6": float(prec_at_06) if prec_at_06 is not None else None,
   245→   245→            "n_high_confidence_test": n_high_conf,
   246→   246→        },
   247→   247→    }

   250→   250→# ---------------------------------------------------------------------------
   251→   251→# Save model artifacts (.pkl + .json)
   252→   252→# ---------------------------------------------------------------------------
   253→   253→def save_model_files(model: xgb.XGBClassifier,
   254→   254→                     regime_name: str,
   255→   255→                     feature_names: list) -> tuple:
   256→   256→    """Save .pkl (sklearn model) and .json (XGBoost native JSON) for one regime.
   257→   257→
   258→   258→    Returns (pkl_path, json_path).
   259→   259→    """
   260→   260→    safe = regime_name.lower()
   261→   261→    pkl_path = OUTPUT_DIR / f"regime_{safe}.pkl"
   262→   262→    json_path = OUTPUT_DIR / f"regime_{safe}.json"
   263→   263→
   264→   264→    # .pkl — full XGBClassifier wrapped in a dict with metadata
   265→   265→    with open(pkl_path, "wb") as f:
   266→   266→        pickle.dump({
   267→   267→            "model": model,
   268→   268→            "feature_names": feature_names,
   269→   269→            "regime": regime_name,
   270→   270→            "long_threshold": LONG_THRESHOLD,
   271→   271→            "short_threshold": SHORT_THRESHOLD,
   272→   272→        }, f)
   273→   273→
   274→   274→    # .json — XGBoost native JSON via booster.save_raw(raw_format='json')
   275→   275→    # This is the format XGBoost itself loads (full model spec including
   276→   276→    # feature_names, learner params, trees). The TS side parses this.
   277→   277→    booster = model.get_booster()
   278→   278→    raw = booster.save_raw(raw_format="json")
   279→   279→    with open(json_path, "wb") as f:
   280→   280→        f.write(raw)
   281→   281→
   282→   282→    return pkl_path, json_path

   286→   286→# Main
   287→   287→# ---------------------------------------------------------------------------
   288→   288→def main():
   289→   289→    parser = argparse.ArgumentParser()
   290→   290→    parser.add_argument("--days", type=int, default=180, help="History depth in days")
   291→   291→    parser.add_argument("--tickers", type=str, default="all",
   292→   292→                        help="Comma-separated tickers (default: all 11)")
   293→   293→    parser.add_argument("--horizon", type=int, default=DEFAULT_HORIZON,
   294→   294→                        help=f"Forward horizon in bars (default {DEFAULT_HORIZON} = 30min)")
   295→   295→    parser.add_argument("--threshold", type=float, default=DEFAULT_THRESHOLD,
   296→   296→                        help=f"Min forward return to label as up (default {DEFAULT_THRESHOLD})")
   297→   297→    parser.add_argument("--min-samples", type=int, default=MIN_SAMPLES_PER_REGIME,
   298→   298→                        help=f"Min samples per regime to train (default {MIN_SAMPLES_PER_REGIME})")
   299→   299→    args = parser.parse_args()
   300→   300→
   301→   301→    tickers = TICKERS if args.tickers == "all" else args.tickers.split(",")
   302→   302→    min_samples = args.min_samples
   303→   303→
   304→   304→    log("=" * 78)
   305→   305→    log("🚀 Regime-Aware ML Trainer V4 — 12 binary XGBoost classifiers (one per regime)")
   306→   306→    log(f"   days={args.days}, tickers={len(tickers)}, horizon={args.horizon} bars "
   307→   307→        f"(~{args.horizon*5}min), threshold={args.threshold}")
   308→   308→    log(f"   min_samples_per_regime={min_samples}, xgb max_depth=3, reg_lambda=10")
   309→   309→    log(f"   decision rule: P(up)>{LONG_THRESHOLD} → LONG, P(up)<{SHORT_THRESHOLD} → SHORT, else FLAT")
   310→   310→    log("=" * 78)
   311→   311→
   312→   312→    # === Step 1: Load all data ===
   313→   313→    log("\n[Step 1/5] Loading multi-timeframe data for all tickers...")
   314→   314→    t0 = time.time()
   315→   315→    all_aligned = {}
   316→   316→    for i, ticker in enumerate(tickers):
   317→   317→        log(f"  [{i+1}/{len(tickers)}] {ticker}...")
   318→   318→        try:
   319→   319→            data = download_multi_timeframe(ticker, days=args.days)
   320→   320→            if "5min_close" not in data:
   321→   321→                log(f"    SKIP: no 5min data")
   322→   322→                continue
   323→   323→            aligned = align_timeframes(data)
   324→   324→            all_aligned[ticker] = aligned
   325→   325→        except Exception as e:
   326→   326→            log(f"    ERROR: {e}")
   327→   327→    log(f"  Loaded {len(all_aligned)}/{len(tickers)} tickers in {time.time()-t0:.1f}s")
   328→   328→
   329→   329→    if not all_aligned:
   330→   330→        log("FATAL: no data loaded")
   331→   331→        return 1
   332→   332→
   333→   333→    # === Step 2: Compute features + regime + binary label per ticker ===
   334→   334→    log("\n[Step 2/5] Computing features + regimes + binary labels per ticker...")
   335→   335→    t0 = time.time()
   336→   336→    ticker_arrays = []  # list of (ticker, X, regime, y, n)
   337→   337→    feature_names = None
   338→   338→    for i, (ticker, aligned) in enumerate(all_aligned.items()):
   339→   339→        log(f"  [{i+1}/{len(all_aligned)}] {ticker}: features + regime + label...")
   340→   340→        try:
   341→   341→            X, names = compute_features(aligned)
   342→   342→            if feature_names is None:
   343→   343→                feature_names = list(names)
   344→   344→            elif list(names) != feature_names:
   345→   345→                log(f"    ⚠️ feature mismatch: got {len(names)}, expected {len(feature_names)} — using intersection")
   346→   346→                common = sorted(set(names) & set(feature_names))
   347→   347→                if not common:
   348→   348→                    log(f"    SKIP: no common features with previous tickers")
   349→   349→                    continue
   350→   350→                old_idx = {n: j for j, n in enumerate(names)}
   351→   351→                new_idx = {n: j for j, n in enumerate(feature_names)}
   352→   352→                # Remap current ticker's X to common features order
   353→   353→                X_new = np.zeros((len(X), len(common)))
   354→   354→                for j, cn in enumerate(common):
   355→   355→                    X_new[:, j] = X[:, old_idx[cn]]
   356→   356→                X = X_new
   357→   357→                feature_names = common
   358→   358→                # Remap previous tickers' X to common features order
   359→   359→                new_arrays = []
   360→   360→                for t, X_t, reg_t, y_t, n_t in ticker_arrays:
   361→   361→                    X_t_new = np.zeros((len(X_t), len(common)))
   362→   362→                    # We didn't save the per-ticker feature_names order before.
   363→   363→                    # Since align_timeframes always fills higher TFs, previous tickers
   364→   364→                    # should have used the same feature_names as the first ticker.
   365→   365→                    # If we're here, something is off — but assume the first ticker's
   366→   366→                    # features matched `feature_names` (pre-shrink) and remap by position.
   367→   367→                    if X_t.shape[1] >= len(common):
   368→   368→                        # Best-effort: drop extra columns from the end
   369→   369→                        X_t_new = X_t[:, :len(common)]
   370→   370→                    else:
   371→   371→                        X_t_new[:, :X_t.shape[1]] = X_t
   372→   372→                    new_arrays.append((t, X_t_new, reg_t, y_t, n_t))
   373→   373→                ticker_arrays = new_arrays
   374→   374→
   375→   375→            close5 = aligned["5min_close"]
   376→   376→            high5 = aligned["5min_high"]
   377→   377→            low5 = aligned["5min_low"]
   378→   378→            open5 = aligned["5min_open"]
   379→   379→            vol5 = aligned["5min_volume"]
   380→   380→
   381→   381→            # Regime via meta_labeler_v2.compute_regime_v2 (uses precompute_indicators)
   382→   382→            ind = precompute_indicators(open5, close5, high5, low5, vol5)
   383→   383→            regime = compute_regime_v2(close5, high5, low5, ind)
   384→   384→
   385→   385→            # Binary label: y = 1 if forward_return > threshold else 0; -1 = drop
   386→   386→            y = compute_binary_label(close5, horizon=args.horizon, threshold=args.threshold)
   387→   387→
   388→   388→            ticker_arrays.append((ticker, X, regime, y, len(X)))
   389→   389→
   390→   390→            # Stats
   391→   391→            valid = y >= 0
   392→   392→            log(f"    X={X.shape}, regimes computed, y positive_rate (valid bars)="
   393→   393→                f"{y[valid].sum()}/{valid.sum()} = {y[valid].mean() if valid.sum() else 0:.3f}")
   394→   394→            # Per-regime count for this ticker
   395→   395→            r_counts = {REGIME_NAMES[r]: int((regime[valid] == r).sum()) for r in range(len(REGIME_NAMES))}
   396→   396→            log(f"    regime counts: " + ", ".join(f"{k}={v}" for k, v in r_counts.items() if v > 0))
   397→   397→        except Exception as e:
