   130→/**
   131→ * Simple causal rolling mean via cumsum.
   132→ *   result[i < w]  = cumsum[i] / w   (matches numpy rolling_mean with cumsum trick:
   133→ *                                   warmup gets partial sum / w, NOT / (i+1))
   134→ *   result[i >= w] = (cumsum[i] - cumsum[i-w]) / w
   135→ *
   136→ * Identical to the `rolling_mean()` helper in ml_features.py.
   137→ */
   138→function rollingMean(arr: number[], w: number): number[] {
   139→  const n = arr.length
   140→  const result = new Array<number>(n).fill(0)
   141→  if (n === 0) return result
   142→  const cumsum = new Array<number>(n)
   143→  let s = 0
   144→  for (let i = 0; i < n; i++) {
   145→    s += arr[i]
   146→    cumsum[i] = s
   147→  }
   148→  for (let i = 0; i < n; i++) {
   149→    if (i < w) {
   150→      // Python rolling_mean returns cumsum[i] / w (smaller than true mean during warmup).
   151→      result[i] = cumsum[i] / w
   152→    } else {
   153→      result[i] = (cumsum[i] - cumsum[i - w]) / w
   154→    }
   155→  }
   156→  return result
   157→}
   158→
   159→/**
   160→ * Causal SMA used in ml_features.py `causal_sma()` — DIFFERENT from rolling_mean:
   161→ *   result[i < w]  = cumsum[i] / (i + 1)   (true partial mean)
   162→ *   result[i >= w] = (cumsum[i] - cumsum[i-w]) / w
   163→ *
   164→ * Used for sma5/14/20/50.
   165→ */
   166→function causalSMA(arr: number[], w: number): number[] {
   167→  const n = arr.length
   168→  const result = new Array<number>(n).fill(0)
   169→  if (n === 0) return result
   170→  const cumsum = new Array<number>(n)
   171→  let s = 0
   172→  for (let i = 0; i < n; i++) {
   173→    s += arr[i]
   174→    cumsum[i] = s
   175→  }
   176→  for (let i = 0; i < n; i++) {
   177→    if (i < w) {
   178→      result[i] = cumsum[i] / (i + 1)
   179→    } else {
   180→      result[i] = (cumsum[i] - cumsum[i - w]) / w
   181→    }
   182→  }
   183→  return result
   184→}

   186→/** Standard EMA with smoothing factor k = 2/(period+1). Matches `ema()` in ml_features.py. */
   187→function ema(arr: number[], period: number): number[] {
   188→  const n = arr.length
   189→  const result = new Array<number>(n).fill(0)
   190→  if (n === 0) return result
   191→  const k = 2 / (period + 1)
   192→  result[0] = arr[0]
   193→  for (let i = 1; i < n; i++) {
   194→    result[i] = arr[i] * k + result[i - 1] * (1 - k)
   195→  }
   196→  return result
   197→}

   199→/**
   200→ * Compute the 31-feature vector at the LAST bar of `candles`.
   201→ *
   202→ * Returns a fixed-length array of 31 numbers, in alphabetical feature order.
   203→ * Mirrors `compute_features()` in ml_features.py exactly.
   204→ *
   205→ * Higher-timeframe features (1h_*, 1d_*) are approximated from 5min candles
   206→ * using the same approach as the existing v2 meta_selector.ts, since the
   207→ * trader server only provides a single 5min candle stream:
   208→ *   - 1h_ret:  (close[i] - close[i-12]) / close[i-12]            (12 × 5min = 60min)
   209→ *   - 1d_ret:  (close[i] - close[i-144]) / close[i-144]          (~1 trading day)
   210→ *   - 1h_trend: close[i] / sma(close, 120)[i]                    (~10 hours of 5min)
   211→ *   - 1d_trend: close[i] / sma(close, 144)[i]                    (~1 trading day)
   212→ *   - 1h_rsi:   rsi14(close)[i]                                  (5min RSI as 1h RSI proxy)
   213→ *
   214→ * The first 50 rows of the training data were zeroed (X[:50] = 0), so the
   215→ * model handles "not enough history" gracefully.  We mirror that by returning
   216→ * all-zeros if n < 50.
   217→ */
   218→export function computeMetaV4Features(candles: Candle[]): number[] {
   219→  const n = candles.length
   220→  const N_FEATURES = 31
   221→  if (n < 50) {
   222→    return new Array<number>(N_FEATURES).fill(0)
   223→  }
   224→
   225→  const close = new Array<number>(n)
   226→  const high = new Array<number>(n)
   227→  const low = new Array<number>(n)
   228→  const vol = new Array<number>(n)
   229→  const time = new Array<number>(n)
   230→  for (let i = 0; i < n; i++) {
   231→    close[i] = candles[i].close
   232→    high[i] = candles[i].high
   233→    low[i] = candles[i].low
   234→    vol[i] = candles[i].volume
   235→    time[i] = candles[i].time
   236→  }
   237→  const idx = n - 1
   238→
   239→  // ── Returns (causal, no wraparound) ──────────────────────────────────────────
   240→  const ret1 = new Array<number>(n).fill(0)
   241→  const ret5 = new Array<number>(n).fill(0)
   242→  const ret10 = new Array<number>(n).fill(0)
   243→  const ret30 = new Array<number>(n).fill(0)
   244→  const ret5log = new Array<number>(n).fill(0)
   245→  for (let i = 1; i < n; i++) {
   246→    ret1[i] = (close[i] - close[i - 1]) / (close[i - 1] + EPS)
   247→  }
   248→  for (let i = 5; i < n; i++) {
   249→    ret5[i] = (close[i] - close[i - 5]) / (close[i - 5] + EPS)
   250→    ret5log[i] = Math.log(close[i] / (close[i - 5] + EPS))
   251→  }
   252→  for (let i = 10; i < n; i++) {
   253→    ret10[i] = (close[i] - close[i - 10]) / (close[i - 10] + EPS)
   254→  }
   255→  for (let i = 30; i < n; i++) {
   256→    ret30[i] = (close[i] - close[i - 30]) / (close[i - 30] + EPS)
   257→  }
   258→
   259→  // ── SMAs ───────────────────────────────────────────────────────────────────
   260→  const sma5 = causalSMA(close, 5)
   261→  const sma14 = causalSMA(close, 14)
   262→  const sma20 = causalSMA(close, 20)
   263→  const sma50 = causalSMA(close, 50)
   264→  // Approx SMAs for higher-TF features
   265→  const sma120 = causalSMA(close, 120)  // ~10 hours of 5min candles
   266→  const sma144 = causalSMA(close, 144)  // ~1 trading day of 5min candles
   267→
   268→  const sma5_sma14 = sma5.map((v, i) => v / (sma14[i] + EPS))
   269→  const sma14_sma20 = sma14.map((v, i) => v / (sma20[i] + EPS))
   270→  const sma20_sma50 = sma20.map((v, i) => v / (sma50[i] + EPS))
   271→  const price_sma20 = close.map((v, i) => v / (sma20[i] + EPS))
   272→  const price_sma50 = close.map((v, i) => v / (sma50[i] + EPS))
   273→
   274→  // ── RSI (SIMPLE rolling mean of gains/losses, NOT Wilder) ─────────────────────
   275→  const deltas = new Array<number>(n)
   276→  deltas[0] = 0
   277→  for (let i = 1; i < n; i++) {
   278→    deltas[i] = close[i] - close[i - 1]
   279→  }
   280→  const gains = new Array<number>(n)
   281→  const losses = new Array<number>(n)
   282→  for (let i = 0; i < n; i++) {
   283→    const d = deltas[i]
   284→    gains[i] = d > 0 ? d : 0
   285→    losses[i] = d < 0 ? -d : 0
   286→  }
   287→  const avgGain14 = rollingMean(gains, 14)
   288→  const avgLoss14 = rollingMean(losses, 14)
   289→  const rsi14 = new Array<number>(n)
   290→  for (let i = 0; i < n; i++) {
   291→    const rs = avgGain14[i] / (avgLoss14[i] + EPS)
   292→    rsi14[i] = 100 - 100 / (1 + rs)
   293→  }
   294→
   295→  // RSI(2) — Connors
   296→  const avgGain2 = rollingMean(gains, 2)
   297→  const avgLoss2 = rollingMean(losses, 2)
   298→  const rsi2 = new Array<number>(n)
   299→  for (let i = 0; i < n; i++) {
   300→    const rs = avgGain2[i] / (avgLoss2[i] + EPS)
   301→    rsi2[i] = 100 - 100 / (1 + rs)
   302→  }
   303→
   304→  // ── Bollinger Bands ─────────────────────────────────────────────────────────
   305→  // Python: std20[i] = np.std(close5[max(0,i-19):i+1], ddof=1) if i>=19 else 0
   306→  //       = sample stddev over last 20 closes (inclusive of i)
   307→  const std20 = new Array<number>(n).fill(0)
   308→  for (let i = 19; i < n; i++) {
   309→    let mean = 0
   310→    for (let j = i - 19; j <= i; j++) mean += close[j]
   311→    mean /= 20
   312→    let sq = 0
   313→    for (let j = i - 19; j <= i; j++) {
   314→      const d = close[j] - mean
   315→      sq += d * d
   316→    }
   317→    // ddof=1 → divide by (n-1)=19
   318→    std20[i] = Math.sqrt(sq / 19)
   319→  }
   320→  const bbUpper = new Array<number>(n)
   321→  const bbLower = new Array<number>(n)
   322→  const bbWidth = new Array<number>(n)
   323→  const bbPctB = new Array<number>(n)
   324→  for (let i = 0; i < n; i++) {
   325→    bbUpper[i] = sma20[i] + 2 * std20[i]
   326→    bbLower[i] = sma20[i] - 2 * std20[i]
   327→    bbWidth[i] = (4 * std20[i]) / (sma20[i] + EPS)
   328→    bbPctB[i] = (close[i] - bbLower[i]) / (4 * std20[i] + EPS)
   329→  }
   330→  const price_bb_upper = close.map((v, i) => v / (bbUpper[i] + EPS))
   331→  const price_bb_lower = close.map((v, i) => v / (bbLower[i] + EPS))
   332→
   333→  // ── MACD ─────────────────────────────────────────────────────────────────────
   334→  const ema12 = ema(close, 12)
   335→  const ema26 = ema(close, 26)
   336→  const macdLine = new Array<number>(n)
   337→  for (let i = 0; i < n; i++) macdLine[i] = ema12[i] - ema26[i]
   338→  const macdSignal = ema(macdLine, 9)
   339→  const macdHist = new Array<number>(n)
   340→  for (let i = 0; i < n; i++) macdHist[i] = macdLine[i] - macdSignal[i]
   341→  // Normalize by close (matches ml_features.py)
   342→  const macd_line = macdLine.map((v, i) => v / (close[i] + EPS))
   343→  const macd_signal = macdSignal.map((v, i) => v / (close[i] + EPS))
   344→
   345→  // ── ATR (SIMPLE rolling mean of TR, NOT Wilder) ──────────────────────────────
   346→  // Python: tr = max(h-l, |h-prev_close|, |l-prev_close|) with np.roll(close, 1).
   347→  // np.roll wraps (last→first), but first bar is in warmup. We use causal
   348→  // prev_close (close[i-1] for i>0, close[0] for i=0) which matches Python
   349→  // for i>=1 exactly.
   350→  const tr = new Array<number>(n)
   351→  for (let i = 0; i < n; i++) {
   352→    const prevClose = i === 0 ? close[0] : close[i - 1]
   353→    const a = high[i] - low[i]
   354→    const b = Math.abs(high[i] - prevClose)
   355→    const c = Math.abs(low[i] - prevClose)
   356→    tr[i] = Math.max(a, Math.max(b, c))
   357→  }
   358→  const atr14 = rollingMean(tr, 14)
   359→  const atr_pct = atr14.map((v, i) => v / (close[i] + EPS))
   360→
   361→  // ── Volume ratio ─────────────────────────────────────────────────────────────
   362→  const volAvg20 = rollingMean(vol, 20)
   363→  const vol_ratio = vol.map((v, i) => v / (volAvg20[i] + EPS))
   364→
   365→  // ── OBV slope (10-bar window, NOT 5) ─────────────────────────────────────────
   366→  const obv = new Array<number>(n)
   367→  obv[0] = 0
   368→  for (let i = 1; i < n; i++) {
   369→    obv[i] = obv[i - 1] + (deltas[i] > 0 ? vol[i] : -vol[i])
   370→  }
   371→  const obv_slope = new Array<number>(n).fill(0)
   372→  for (let i = 10; i < n; i++) {
   373→    obv_slope[i] = (obv[i] - obv[i - 10]) / (obv[i - 10] + EPS)
   374→  }
   375→
   376→  // ── Stochastic %K (14-bar high/low, INCLUSIVE of current bar) ────────────────
   377→  // Python: hh14[i] = max(high[max(0,i-13):i+1]); ll14[i] = min(low[...])
   378→  const stoch_k = new Array<number>(n)
   379→  for (let i = 0; i < n; i++) {
   380→    const start = Math.max(0, i - 13)
   381→    let hh = -Infinity
   382→    let ll = Infinity
   383→    for (let j = start; j <= i; j++) {
   384→      if (high[j] > hh) hh = high[j]
   385→      if (low[j] < ll) ll = low[j]
   386→    }
   387→    stoch_k[i] = ((close[i] - ll) / (hh - ll + EPS)) * 100
   388→  }
   389→
   390→  // ── ADX (SIMPLIFIED — NOT Wilder) ───────────────────────────────────────────
   391→  // Python:
   392→  //   up_moves   = where(deltas > 0, 1, 0)
   393→  //   down_moves = where(deltas < 0, 1, 0)
   394→  //   adx = abs(rolling_mean(up_moves, 14) - rolling_mean(down_moves, 14)) * 100
   395→  const up_moves = new Array<number>(n)
   396→  const down_moves = new Array<number>(n)
   397→  for (let i = 0; i < n; i++) {
   398→    up_moves[i] = deltas[i] > 0 ? 1 : 0
   399→    down_moves[i] = deltas[i] < 0 ? 1 : 0
   400→  }
   401→  const rm_up = rollingMean(up_moves, 14)
   402→  const rm_dn = rollingMean(down_moves, 14)
   403→  const adx = new Array<number>(n)
   404→  for (let i = 0; i < n; i++) {
   405→    adx[i] = Math.abs(rm_up[i] - rm_dn[i]) * 100
   406→  }
   407→
   408→  // ── Higher timeframe context (approximations from 5min candles) ──────────────
   409→  // ml_features.py uses real 1hour and 1day candle series from the multi-TF
   410→  // data pipeline.  The trader server only has 5min candles, so we approximate.
   411→  // This matches the existing v2 meta_selector.ts approach (tested & deployed).
   412→  const ret1h = idx >= 12 ? (close[idx] - close[idx - 12]) / (close[idx - 12] + EPS) : 0
   413→  const ret1d = idx >= 144 ? (close[idx] - close[idx - 144]) / (close[idx - 144] + EPS) : 0
   414→  // 1h_trend: close / sma10(close1h)  ≈ close / sma120(close5min)  [10 hours]
   415→  const trend1h = close[idx] / (sma120[idx] + EPS)
   416→  // 1d_trend: close / sma5(close1d)   ≈ close / sma144(close5min)  [~1 trading day]
   417→  const trend1d = close[idx] / (sma144[idx] + EPS)
   418→  // 1h_rsi: approximation — use 5min RSI14 (no real 1hour close series available)
   419→  const rsi1h = rsi14[idx]
   420→
   421→  // ── Time features (MSK timezone, UTC+3) ──────────────────────────────────────
   422→  // ml_features.py:
   423→  //   ts_seconds = time5 / 1000                    # time5 is in ms
   424→  //   hours = (ts_seconds // 3600 + 3) % 24        # MSK = UTC+3
   425→  //   dow   = (ts_seconds // 86400 + 4) % 7        # Sun=0...Sat=6 (matches JS getUTCDay)
   426→  //   hour         = hours / 24.0
   427→  //   day_of_week  = dow / 7.0
   428→  //
   429→  // regime_detector.ts Candle.time is "unix seconds (or any monotonic unit)".
   430→  // The trader server's Candle.time is in MILLISECONDS (existing meta_selector.ts
   431→  // does `new Date(ts)` directly, which expects ms).  We auto-detect: if ts > 1e12
   432→  // (year 2001+), treat as ms; else seconds.
   433→  const ts = time[idx]
   434→  const tsMs = ts > 1e12 ? ts : ts * 1000
   435→  const date = new Date(tsMs)
   436→  const hour = ((date.getUTCHours() + 3) % 24) / 24.0
   437→  const day_of_week = date.getUTCDay() / 7.0
   438→
   439→  // ── Assemble feature vector in ALPHABETICAL order ────────────────────────────
   440→  // (matches training metadata feature_names exactly)
   441→  const features: Record<string, number> = {
   442→    '1d_ret': ret1d,
   443→    '1d_trend': trend1d,
   444→    '1h_ret': ret1h,
   445→    '1h_rsi': rsi1h,
   446→    '1h_trend': trend1h,
   447→    'adx': adx[idx],
   448→    'atr_pct': atr_pct[idx],
   449→    'bb_pct_b': bbPctB[idx],
   450→    'bb_width': bbWidth[idx],
   451→    'day_of_week': day_of_week,
   452→    'hour': hour,
   453→    'macd_hist': macdHist[idx],
   454→    'macd_line': macd_line[idx],
   455→    'macd_signal': macd_signal[idx],
   456→    'obv_slope': obv_slope[idx],
   457→    'price_bb_lower': price_bb_lower[idx],
   458→    'price_bb_upper': price_bb_upper[idx],
   459→    'price_sma20': price_sma20[idx],
   460→    'price_sma50': price_sma50[idx],
   461→    'ret_1': ret1[idx],
   462→    'ret_10': ret10[idx],
   463→    'ret_30': ret30[idx],
   464→    'ret_5': ret5[idx],
   465→    'ret_5_log': ret5log[idx],
   466→    'rsi14': rsi14[idx],
   467→    'rsi2': rsi2[idx],
   468→    'sma14_sma20': sma14_sma20[idx],
   469→    'sma20_sma50': sma20_sma50[idx],
   470→    'sma5_sma14': sma5_sma14[idx],
   471→    'stoch_k': stoch_k[idx],
   472→    'vol_ratio': vol_ratio[idx],
   473→  }
   474→
   475→  // Build the vector in the exact order specified by the task / training metadata.
   476→  const FEATURE_NAMES = [
   477→    '1d_ret', '1d_trend', '1h_ret', '1h_rsi', '1h_trend', 'adx', 'atr_pct',
   478→    'bb_pct_b', 'bb_width', 'day_of_week', 'hour', 'macd_hist', 'macd_line',
   479→    'macd_signal', 'obv_slope', 'price_bb_lower', 'price_bb_upper',
   480→    'price_sma20', 'price_sma50', 'ret_1', 'ret_10', 'ret_30', 'ret_5',
   481→    'ret_5_log', 'rsi14', 'rsi2', 'sma14_sma20', 'sma20_sma50', 'sma5_sma14',
   482→    'stoch_k', 'vol_ratio',
   483→  ]
   484→  const out = new Array<number>(FEATURE_NAMES.length)
   485→  for (let i = 0; i < FEATURE_NAMES.length; i++) {
   486→    let v = features[FEATURE_NAMES[i]]
   487→    if (!Number.isFinite(v)) v = 0
   488→    // Python: np.clip(X, -10, 10)
   489→    if (v > 10) v = 10
   490→    else if (v < -10) v = -10
   491→    out[i] = v
   492→  }
   493→  return out
   494→}

   496→// ─── Model loader (with path search + caching) ───────────────────────────────
   497→
   498→/**
   499→ * Resolve the absolute path of a regime model file.
   500→ *
   501→ * Tries, in order:
   502→ *   /opt/ai-trader/data/regime_<name>.json
   503→ *   /opt/ai-trader/src/strategies/regime_<name>.json
   504→ *   <__dirname>/regime_<name>.json
   505→ *   <cwd>/regime_<name>.json
   506→ *
   507→ * Returns the first existing path, or null if none found.
   508→ */
   509→function resolveModelPath(modelFile: string): string | null {
   510→  for (const dir of MODEL_SEARCH_PATHS) {
   511→    const p = path.join(dir, modelFile)
   512→    try {
   513→      if (fs.existsSync(p)) return p
   514→    } catch {
   515→      // ignore — try next
   516→    }
   517→  }
   518→  return null
   519→}
   520→
   521→/** Per-regime model cache: regimeIndex → loaded model (loaded lazily on first hit). */
   522→const _modelByRegime = new Map<number, ReturnType<typeof loadModel> | null>()
   523→
   524→/**
   525→ * Get the (cached) ML model for a regime.
   526→ *
   527→ * Returns null if:
   528→ *   - The regime has no model file (OVERSOLD_BOUNCE, OVERBOUGHT_REVERSAL)
   529→ *   - The model file cannot be found in any search path
   530→ *
   531→ * On first call for a regime, the model is loaded from disk and cached.
   532→ * Subsequent calls return the cached instance without disk I/O.
   533→ */
   534→function getRegimeModel(regimeIdx: number): ReturnType<typeof loadModel> | null {
   535→  if (_modelByRegime.has(regimeIdx)) {
   536→    return _modelByRegime.get(regimeIdx) ?? null
   537→  }
   538→  const modelFile = REGIME_MODEL_FILES[regimeIdx]
   539→  if (!modelFile) {
   540→    _modelByRegime.set(regimeIdx, null)
   541→    return null
   542→  }
   543→  const modelPath = resolveModelPath(modelFile)
   544→  if (!modelPath) {
   545→    console.warn(
   546→      `[MetaSelectorV4] WARN: model file not found: ${modelFile} ` +
   547→        `(searched: ${MODEL_SEARCH_PATHS.join(', ')})`,
   548→    )
   549→    _modelByRegime.set(regimeIdx, null)
   550→    return null
   551→  }
   552→  try {
   553→    const model = loadModel(modelPath)
   554→    _modelByRegime.set(regimeIdx, model)
   555→    console.log(
   556→      `[MetaSelectorV4] loaded ${modelFile}: nTrees=${model.nTrees} nFeatures=${model.nFeatures}`,
   557→    )
   558→    return model
   559→  } catch (e: any) {
   560→    console.error(`[MetaSelectorV4] ERROR loading ${modelFile}: ${e.message}`)
   561→    _modelByRegime.set(regimeIdx, null)
   562→    return null
   563→  }
   564→}
   565→
   566→/** Clear all cached models (useful for hot-reload / testing). */
   567→export function clearMetaV4ModelCache(): void {
   568→  _modelByRegime.clear()
   569→}

   571→// ─── Strategy class ────────────────────────────────────────────────────────────
   572→
   573→/**
   574→ * MetaSelectorV4Strategy — the v4 ML orchestrator.
   575→ *
   576→ *   detectRegime → loadModel → predict_proba → LONG / SHORT / FLAT
   577→ *
   578→ * Fallback for regimes with no ML model (skipped during training due to
   579→ * insufficient samples):
   580→ *   OVERSOLD_BOUNCE     → LONG  (buy the dip — RSI<25 + close>SMA5)
   581→ *   OVERBOUGHT_REVERSAL → SHORT (sell the top — RSI>75 + close<SMA5)
   582→ */
   583→export class MetaSelectorV4Strategy implements IStrategy {
   584→  name = 'meta_selector_v4'
   585→  description =
   586→    'ML v4 orchestrator: detectRegime → per-regime binary XGBoost (P(up)) → LONG/SHORT/FLAT'
   587→
   588→  /** Last prediction details — exposed for logging/monitoring. */
   589→  lastRegimeIdx: number = -1
   590→  lastRegimeName: string = ''
   591→  lastProba: number = 0.5
   592→  lastDecision: number = 0
   593→  lastDecisionLabel: string = 'FLAT'
   594→  lastModelUsed: string = ''
   595→  lastError: string = ''
   596→
   597→  predict(
   598→    candles: Candle[],
   599→    idx: number,
   600→    _hasPosition: boolean,
   601→    _stepsHeld?: number,
   602→    _ctx?: StrategyContext,
   603→  ): number {
   604→    // Reset transient state
   605→    this.lastError = ''
   606→
   607→    // Need enough candle history.  detectRegime requires >100 bars; features
   608→    // need >50; we require 200 for the higher-TF approximations (144-bar SMA).
   609→    if (!candles || candles.length < MIN_CANDLES) {
   610→      this.lastError = `insufficient_history (need >= ${MIN_CANDLES} candles, got ${candles?.length ?? 0})`
   611→      // During warmup, stay flat.
   612→      this.lastDecision = 0
   613→      this.lastDecisionLabel = 'FLAT'
   614→      return 0
   615→    }
   616→
   617→    // 1. Detect regime at the latest bar
   618→    const regime = detectRegime(candles)
   619→    this.lastRegimeIdx = regime.regime
   620→    this.lastRegimeName = regime.regimeName
   621→
   622→    // 2. If regime has no ML model → use rule-based fallback
   623→    const model = getRegimeModel(regime.regime)
   624→    if (!model) {
   625→      let decision: number
   626→      let label: string
   627→      if (regime.regime === 7 /* OVERSOLD_BOUNCE */) {
   628→        decision = 1 // LONG — buy the dip
   629→        label = 'LONG'
   630→      } else if (regime.regime === 8 /* OVERBOUGHT_REVERSAL */) {
   631→        decision = 2 // SHORT — sell the top
   632→        label = 'SHORT'
   633→      } else {
   634→        // Model file not found for a regime that should have one.  Stay flat
   635→        // and log loudly so the operator notices.
   636→        decision = 0
   637→        label = 'FLAT'
   638→        this.lastError = `model_missing_for_${regime.regimeName}`
   639→      }
   640→      this.lastProba = 0.5
   641→      this.lastDecision = decision
   642→      this.lastDecisionLabel = label
   643→      this.lastModelUsed = '(fallback)'
   644→      console.log(
   645→        `[MetaSelectorV4] regime=${regime.regimeName} (idx=${regime.regime}) ` +
   646→          `P(up)=N/A → ${label} [FALLBACK]`,
   647→      )
   648→      return decision
   649→    }
   650→
   651→    // 3. Has ML model → compute features + predict
   652→    this.lastModelUsed = regime.regimeName
   653→    let features: number[]
   654→    try {
   655→      features = computeMetaV4Features(candles)
   656→    } catch (e: any) {
   657→      this.lastError = `feature_error: ${e.message}`
   658→      this.lastDecision = 0
   659→      this.lastDecisionLabel = 'FLAT'
   660→      console.error(`[MetaSelectorV4] feature computation failed: ${e.message}`)
   661→      return 0
   662→    }
   663→
   664→    let proba: number
   665→    try {
   666→      proba = predict_proba(model, features)
   667→    } catch (e: any) {
   668→      this.lastError = `predict_error: ${e.message}`
   669→      this.lastDecision = 0
   670→      this.lastDecisionLabel = 'FLAT'
   671→      console.error(`[MetaSelectorV4] predict_proba failed: ${e.message}`)
   672→      return 0
   673→    }
   674→
   675→    // Guard against NaN/Inf
   676→    if (!Number.isFinite(proba)) {
   677→      proba = 0.5
   678→    }
   679→    this.lastProba = proba
   680→
   681→    // 4. Decision thresholds (from v4 training metadata)
   682→    let decision: number
   683→    let label: string
   684→    if (proba > LONG_THRESHOLD) {
   685→      decision = 1 // LONG
   686→      label = 'LONG'
   687→    } else if (proba < SHORT_THRESHOLD) {
   688→      decision = 2 // SHORT
   689→      label = 'SHORT'
   690→    } else {
   691→      decision = 0 // FLAT
   692→      label = 'FLAT'
   693→    }
   694→    this.lastDecision = decision
   695→    this.lastDecisionLabel = label
   696→
   697→    // 5. Log every prediction
   698→    console.log(
   699→      `[MetaSelectorV4] regime=${regime.regimeName} (idx=${regime.regime}) ` +
   700→        `P(up)=${proba.toFixed(4)} → ${label}`,
   701→    )
   702→
   703→    return decision
   704→  }
   705→}
   706→
   707→/** Factory — matches the trader server's strategy-instantiation pattern. */
   708→export function createMetaSelectorV4Strategy(): MetaSelectorV4Strategy {
   709→  return new MetaSelectorV4Strategy()
